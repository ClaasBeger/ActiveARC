def components(grid: Grid, diagonal: bool=False) -> list[Component]:
    """Return connected non-black same-color components.

    Args:
      grid (Grid): Grid to inspect.
      diagonal (bool): Whether diagonal contact should connect components.

    Returns:
      list[Component]: Connected components.
    """
    height, width = (len(grid), len(grid[0]))
    seen: set[Pixel] = set()
    connected_components: list[Component] = []
    for row in range(height):
        for col in range(width):
            color = grid[row][col]
            if color == 0 or (row, col) in seen:
                continue
            queue = [(row, col)]
            seen.add((row, col))
            pixels: list[Pixel] = []
            while queue:
                pixel = queue.pop()
                pixels.append(pixel)
                neighbors = neighbors8(pixel, height, width) if diagonal else neighbors4(pixel, height, width)
                for neighbor in neighbors:
                    if neighbor in seen:
                        continue
                    if grid[neighbor[0]][neighbor[1]] != color:
                        continue
                    seen.add(neighbor)
                    queue.append(neighbor)
            connected_components.append(Component(color, pixels))
    return connected_components

def neighbors4(pixel: Pixel, height: int, width: int) -> list[Pixel]:
    """Return 4-neighbors within bounds.

    Args:
      pixel (Pixel): Source pixel.
      height (int): Grid height.
      width (int): Grid width.

    Returns:
      list[Pixel]: Neighbor pixels.
    """
    row, col = pixel
    candidates = [(row - 1, col), (row + 1, col), (row, col - 1), (row, col + 1)]
    return [(next_row, next_col) for next_row, next_col in candidates if 0 <= next_row < height and 0 <= next_col < width]

def neighbors8(pixel: Pixel, height: int, width: int) -> list[Pixel]:
    """Return 8-neighbors within bounds.

    Args:
      pixel (Pixel): Source pixel.
      height (int): Grid height.
      width (int): Grid width.

    Returns:
      list[Pixel]: Neighbor pixels.
    """
    row, col = pixel
    return [(next_row, next_col) for next_row in range(row - 1, row + 2) for next_col in range(col - 1, col + 2) if (next_row, next_col) != pixel and 0 <= next_row < height and (0 <= next_col < width)]

def hull_mask(grid: Grid, pixels: Sequence[Pixel]) -> set[Pixel]:
    """Return grid cells inside the convex hull of pixels.

    Args:
      grid (Grid): Reference grid.
      pixels (Sequence[Pixel]): Hull source pixels.

    Returns:
      set[Pixel]: Cells inside or on the hull.
    """
    hull = convex_hull(pixels)
    return {pixel for pixel in all_pixels(grid) if point_in_polygon(pixel, hull) or on_polygon_boundary(pixel, hull)}

def convex_hull(pixels: Sequence[Pixel]) -> list[Pixel]:
    """Return the convex hull of pixels using monotonic chain.

    Args:
      pixels (Sequence[Pixel]): Source pixels.

    Returns:
      list[Pixel]: Hull vertices.
    """
    points = sorted(set(((col, row) for row, col in pixels)))
    if len(points) <= 1:
        return [(row, col) for col, row in points]

    def cross(origin: tuple[int, int], first: tuple[int, int], second: tuple[int, int]) -> int:
        return (first[0] - origin[0]) * (second[1] - origin[1]) - (first[1] - origin[1]) * (second[0] - origin[0])
    lower: list[tuple[int, int]] = []
    for point in points:
        while len(lower) >= 2 and cross(lower[-2], lower[-1], point) <= 0:
            lower.pop()
        lower.append(point)
    upper: list[tuple[int, int]] = []
    for point in reversed(points):
        while len(upper) >= 2 and cross(upper[-2], upper[-1], point) <= 0:
            upper.pop()
        upper.append(point)
    hull = lower[:-1] + upper[:-1]
    return [(row, col) for col, row in hull]

def point_in_polygon(pixel: Pixel, polygon: Sequence[Pixel]) -> bool:
    """Return whether a pixel center lies inside a polygon.

    Args:
      pixel (Pixel): Pixel to test.
      polygon (Sequence[Pixel]): Polygon vertices.

    Returns:
      bool: Whether the pixel lies inside the polygon.
    """
    if len(polygon) < 3:
        return False
    y, x = pixel
    inside = False
    prev_y, prev_x = polygon[-1]
    for next_y, next_x in polygon:
        if (next_y > y) != (prev_y > y):
            x_intersection = (prev_x - next_x) * (y - next_y) / (prev_y - next_y) + next_x
            if x < x_intersection:
                inside = not inside
        prev_y, prev_x = (next_y, next_x)
    return inside

def on_polygon_boundary(pixel: Pixel, polygon: Sequence[Pixel]) -> bool:
    """Return whether a pixel lies on any polygon edge.

    Args:
      pixel (Pixel): Pixel to test.
      polygon (Sequence[Pixel]): Polygon vertices.

    Returns:
      bool: Whether the pixel lies on an edge.
    """
    if len(polygon) < 2:
        return False
    for idx, start in enumerate(polygon):
        end = polygon[(idx + 1) % len(polygon)]
        if between(start, end, pixel):
            return True
    return False

def between(first: Pixel, second: Pixel, candidate: Pixel) -> bool:
    """Return whether ``candidate`` lies on the segment from first to second.

    Args:
      first (Pixel): Segment start.
      second (Pixel): Segment end.
      candidate (Pixel): Pixel to test.

    Returns:
      bool: Whether candidate lies on the segment.
    """
    row1, col1 = first
    row2, col2 = second
    row, col = candidate
    cross = (row - row1) * (col2 - col1) - (col - col1) * (row2 - row1)
    if cross != 0:
        return False
    return min(row1, row2) <= row <= max(row1, row2) and min(col1, col2) <= col <= max(col1, col2)

def between_parallel_lines_mask(grid: Grid, line_pixels: Sequence[Pixel]) -> set[Pixel]:
    """Return cells between simple horizontal or vertical parallel lines.

    Args:
      grid (Grid): Reference grid.
      line_pixels (Sequence[Pixel]): Pixels that define the lines.

    Returns:
      set[Pixel]: Cells between the lines.
    """
    del grid
    rows = [row for row, _ in line_pixels]
    cols = [col for _, col in line_pixels]
    row_values = sorted(set(rows))
    col_values = sorted(set(cols))
    if len(col_values) == 2:
        return {(row, col) for row in range(min(rows), max(rows) + 1) for col in range(col_values[0] + 1, col_values[1])}
    if len(row_values) == 2:
        return {(row, col) for row in range(row_values[0] + 1, row_values[1]) for col in range(min(cols), max(cols) + 1)}
    return set()

def touches_any(pixel: Pixel, pixels: Sequence[Pixel]) -> bool:
    """Return whether a pixel touches any source pixel, including diagonally.

    Args:
      pixel (Pixel): Pixel to test.
      pixels (Sequence[Pixel]): Source pixels.

    Returns:
      bool: Whether the pixel touches the source set.
    """
    source = set(pixels)
    row, col = pixel
    return any(((next_row, next_col) in source for next_row in range(row - 1, row + 2) for next_col in range(col - 1, col + 2) if (next_row, next_col) != pixel))

    """Return all pixel coordinates in a grid.

    Args:

      list[Pixel]: Pixel coordinates.

def pixels_by_color(grid: Grid) -> dict[int, list[Pixel]]:
    """Return non-black pixels grouped by color.

    Args:
      grid (Grid): Grid to inspect.

    Returns:
      dict[int, list[Pixel]]: Pixels keyed by color.
    """
    colors: dict[int, list[Pixel]] = {}
    for row, values in enumerate(grid):
        for col, color in enumerate(values):
            if color == 0:
                continue
            colors.setdefault(color, []).append((row, col))
    return colors

def bounds(pixels: Sequence[Pixel]) -> Bounds:
    """Return row/column bounds for pixels.

    Args:
      pixels (Sequence[Pixel]): Pixels to bound.

    Returns:
      Bounds: ``min_row, min_col, max_row, max_col``.
    """
    rows = [row for row, _ in pixels]
    cols = [col for _, col in pixels]
    return (min(rows), min(cols), max(rows), max(cols))

def scene_objects(grid: Grid, diagonal: bool=True) -> list[SceneObject]:
    """Return connected non-black objects in a grid.

    Args:
      grid (Grid): Grid to inspect.
      diagonal (bool): Whether diagonal contact connects components.

    Returns:
      list[SceneObject]: Scene objects sorted by position and color.
    """
    objects = [SceneObject(comp.color, tuple(sorted(comp.pixels))) for comp in components(grid, diagonal=diagonal)]
    return sorted(objects, key=lambda obj: (object_bounds(obj)[0], object_bounds(obj)[1], obj.color, obj.area))

def object_bounds(obj: SceneObject) -> tuple[int, int, int, int]:
    """Return object bounds.

    Args:
      obj (SceneObject): Object to inspect.

    Returns:
      tuple[int, int, int, int]: ``min_row, min_col, max_row, max_col``.
    """
    return bounds(obj.pixels)

def dominant_color_objects(grid: Grid) -> SceneObjectGroup:
    """Return all connected objects in the dominant non-background color.

    Args:
      grid (Grid): Grid to inspect.

    Returns:
      SceneObjectGroup: Same-color components for the dominant color.
    """
    color_pixels = pixels_by_color(grid)
    if not color_pixels:
        raise ValueError('dominant_color_objects requires non-background pixels')
    dominant_color = max(color_pixels, key=lambda color: len(color_pixels[color]))
    objects = [obj for obj in scene_objects(grid, diagonal=False) if obj.color == dominant_color]
    if not objects:
        raise ValueError('dominant_color_objects found no dominant-color objects')
    return SceneObjectGroup(tuple(objects))

def upper_open_hull_region(grid: Grid, pixels: list[Pixel]) -> set[Pixel]:
    """Return a hull region for shapes with an open upper boundary.

    Args:
      grid (Grid): Grid containing the shape.
      pixels (list[Pixel]): Container pixels.

    Returns:
      set[Pixel]: Hull cells below the upper opening, plus boundary-touching
      cells near the top edge.
    """
    min_row = bounds(pixels)[0]
    return {pixel for pixel in hull_mask(grid, pixels) if pixel[0] > min_row + 1 or touches_any(pixel, pixels)}

def container_relative_objects(grid: Grid, container_selector: dict[str, Any], relation: str, candidate_selector: dict[str, Any] | None=None, allow_empty: bool=False, region: str='interior', exclude_container_colors: bool=False) -> SceneObjectGroup:
    """Return objects inside or outside a selected container object.

    Args:
      grid (Grid): Grid to inspect.
      container_selector (dict[str, Any]): Selector used to choose the
          container object.
      relation (str): Either ``"inside"`` or ``"outside"``.
      candidate_selector (dict[str, Any] | None): Optional selector limiting
          which objects are tested against the container.
      allow_empty (bool): Whether an empty relation result is meaningful.
      region (str): Container-region interpretation.
      exclude_container_colors (bool): Whether to skip candidate objects whose
          color belongs to the container selection.

    Returns:
      SceneObjectGroup: Objects satisfying the requested container relation.
    """
    container_selection = select_scene_objects(container_selector, grid)
    container_region = _container_selection_region(grid, container_selection, region)
    container_pixels = _container_selection_pixels(container_selection)
    excluded_colors = _container_selection_colors(container_selection) if exclude_container_colors else set()
    candidate_objects = _container_relation_candidates(grid, container_pixels, candidate_selector, excluded_colors)
    objects = [obj for obj in candidate_objects if _object_container_relation(obj, container_region)]
    if relation == 'outside':
        objects = [obj for obj in candidate_objects if not _object_container_relation(obj, container_region)]
    elif relation != 'inside':
        raise ValueError('Unknown container relation: ' + relation)
    if not objects:
        if allow_empty:
            return SceneObjectGroup(())
        raise ValueError(relation + '_container_objects selector found no objects')
    return SceneObjectGroup(tuple(objects))

def expect_group(selection: SceneSelection | None) -> SceneObjectGroup:
    """Return a selected object group or raise a type error.

    Args:
      selection (SceneSelection | None): Selection to validate.

    Returns:
      SceneObjectGroup: Selected object group.
    """
    if isinstance(selection, SceneObjectGroup):
        return selection
    if isinstance(selection, SceneObject):
        return SceneObjectGroup((selection,))
    if isinstance(selection, tuple):
        return SceneObjectGroup(selection)
    raise ValueError('DSL action requires an object-group selection')

def _container_relation_candidates(grid: Grid, container_pixels: set[Pixel], candidate_selector: dict[str, Any] | None, excluded_colors: set[int]) -> list[SceneObject]:
    """Return candidate objects for a container-relative selector.

    Args:
      grid (Grid): Grid to inspect.
      container_pixels (set[Pixel]): Pixels occupied by the selected container.
      candidate_selector (dict[str, Any] | None): Optional selector limiting
          candidates.
      excluded_colors (set[int]): Candidate colors to exclude.

    Returns:
      list[SceneObject]: Candidate objects disjoint from the container.
    """
    if candidate_selector is None:
        candidates = SceneObjectGroup(tuple(scene_objects(grid, diagonal=False)))
    else:
        candidates = expect_group(select_scene_objects(candidate_selector, grid))
    return [obj for obj in candidates.objects if obj.color not in excluded_colors and container_pixels.isdisjoint(obj.pixels)]

def _container_selection_region(grid: Grid, selection: SceneSelection, region: str) -> set[Pixel]:
    """Return a containment region for a selected object or object group.

    Args:
      grid (Grid): Grid to inspect.
      selection (SceneSelection): Container object or group.
      region (str): Region interpretation.

    Returns:
      set[Pixel]: Container-region pixels.
    """
    objects = _container_selection_objects(selection)
    pixels = sorted({pixel for obj in objects for pixel in obj.pixels})
    if region == 'between_parallel_lines':
        return between_parallel_lines_mask(grid, pixels)
    if region == 'upper_open_hull':
        return upper_open_hull_region(grid, pixels)
    raise ValueError('Unknown container region mode: ' + region)

    Returns:

def _object_container_relation(obj: SceneObject, region: set[Pixel]) -> bool:
    """Return whether an object lies inside a container region.

    Args:
      obj (SceneObject): Object to test.
      region (set[Pixel]): Container interior pixels.

    Returns:
      bool: Whether every pixel of the object lies inside the region.
    """
    return bool(region) and set(obj.pixels).issubset(region)

def isolate_object_group_same_canvas(grid: Grid, group: SceneObjectGroup) -> Grid:
    """Return a same-size grid containing only a selected object group.

    Args:
      grid (Grid): Input grid.
      group (SceneObjectGroup): Objects to keep.

    Returns:
      Grid: Same-size isolated-group grid.
    """
    output = blank_like(grid)
    for obj in group.objects:
        for row, col in obj.pixels:
            output[row][col] = grid[row][col]
    return output

You are dealing with grid transformation tasks: an input grid is mapped to an output grid according to an underlying rule.
Grids are rectangular matrices with integer colors 0–9.

You are the Teacher in Inverse Query Generation.
You are given one input-output example, the rule in words, and the verifier program that will grade the exam. A Student has none of these.
Your job is to teach the Student using demonstrations and optional probes, then send them to an exam.

Rules:
- show_example: you author both input and output. If it matches the rule, the Student is shown the pair and you only get a success acknowledgement. If it does not match, the Student does not see it and you receive the gold output.
- show_transformed_input: you author only the input; the environment fills the gold output and shows the pair to the Student.
- query_student: the Student predicts an output for your input. You see their answer vs gold. They do NOT see gold unless you later show that pair with show_example or show_transformed_input.
- start_exam: the Student then answers 10 held-out generator inputs one at a time, with no feedback, using only the demonstrations you have shown.
- Your one example is teacher-only. The Student does not see it unless you show that pair with show_example or show_transformed_input.
- The Student never sees the rule or the verifier. Do not try to speak to them in prose; only the tools above change what they see.
- The trial is scored by how many of the 10 exam items the Student gets right. The number of show_* and query_student calls is also logged.
